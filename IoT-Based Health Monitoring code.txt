#include <WiFi.h>
#include <HTTPClient.h>
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <DHT.h>
#include <DHT_U.h>
#include <WebServer.h>

// ====== CONFIG — fill these in locally, NEVER commit real values ======
const char* ssid     = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";
String botToken       = "YOUR_TELEGRAM_BOT_TOKEN";
String chatID          = "YOUR_TELEGRAM_CHAT_ID";
// ========================================================================

// Pin Definitions
const int flexPin1 = 34;
const int flexPin2 = 35;
const int thermistorPin = 32;
const int dhtPin = 27;
const int ledPin1 = 26;
const int ledPin2 = 33;
const int ledPin3 = 14;

// OLED display width and height
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64

// Web Server
WebServer server(80);

// OLED display object connected via I2C
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

// DHT sensor object
DHT dht(dhtPin, DHT22);

// Stores the last pattern received
String lastMatchedPattern = "No demand yet.";

// Sends a message to Telegram via bot API
void sendMessageToTelegram(String message) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    String url = "https://api.telegram.org/bot" + botToken + "/sendMessage?chat_id=" + chatID + "&text=" + message;
    http.begin(url);
    int httpResponseCode = http.GET();
    if (httpResponseCode <= 0) {
      Serial.println("Error on sending request");
    }
    http.end();
  }
}

// Serves the web dashboard
void handleRoot() {
  String html = R"(
    <!DOCTYPE html>
    <html lang='en'>
    <head>
      <meta charset='UTF-8'>
      <meta name='viewport' content='width=device-width, initial-scale=1.0'>
      <title>IoT MedTech</title>
      <style>
        body {
          font-family: 'Arial', sans-serif;
          background-color: #f4f4f9;
          color: #333;
          display: flex;
          flex-direction: column;
          align-items: center;
        }
        h1 {
          background-color: #4CAF50;
          color: white;
          padding: 20px 0;
          text-align: center;
          width: 100%;
        }
        .container {
          margin: 20px auto;
          padding: 20px;
          background-color: white;
          box-shadow: 0 0 10px rgba(0, 0, 0, 0.1);
        }
        .pattern {
          font-size: 1.5em;
          margin: 20px 0;
          padding: 10px;
          text-align: center;
          background-color: #e9ecef;
        }
      </style>
    </head>
    <body>
      <div class='container'>
        <h1>Patient Demand</h1>
        <div id="pattern" class="pattern">No demand yet.</div>
      </div>
      <script>
        function updatePattern(pattern) {
          document.getElementById('pattern').innerHTML = 'Demand: ' + pattern;
          speakPattern(pattern);
        }

        function speakPattern(pattern) {
          if ('speechSynthesis' in window) {
            var utterance = new SpeechSynthesisUtterance(pattern);
            window.speechSynthesis.cancel();
            window.speechSynthesis.speak(utterance);
          } else {
            console.error('Speech synthesis not supported.');
          }
        }

        function fetchPattern() {
          fetch('/matched-pattern')
            .then(response => response.text())
            .then(data => {
              updatePattern(data);
            })
            .catch(error => console.error('Error fetching pattern data:', error));
        }

        setInterval(fetchPattern, 3000);
      </script>
    </body>
    </html>
  )";

  server.send(200, "text/html", html);
}

// Returns the last matched demand pattern
void handlePattern() {
  server.send(200, "text/plain", lastMatchedPattern);
}

void setup() {
  Serial.begin(115200);

  WiFi.begin(ssid, password);
  Serial.print("Connecting to Wi-Fi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println();
  Serial.println("Connected to Wi-Fi");
  Serial.print("ESP32 IP address: ");
  Serial.println(WiFi.localIP());

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println(F("SSD1306 allocation failed"));
    for (;;);
  }
  display.clearDisplay();

  dht.begin();

  server.on("/", handleRoot);
  server.on("/matched-pattern", handlePattern);
  server.begin();
  Serial.println("Web server started.");
}

void loop() {
  server.handleClient();

  int flexValue1 = analogRead(flexPin1);
  int flexValue2 = analogRead(flexPin2);
  float humidity = dht.readHumidity();
  float temperature = dht.readTemperature();

  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);
  display.print("Temp: ");
  display.print(temperature);
  display.println(" C");
  display.print("Humidity: ");
  display.print(humidity);
  display.println(" %");

  if (flexValue1 < 4095 && flexValue2 < 4095) {
    Serial.println("HELP");
    display.setCursor(0, 40);
    display.println("HELP");
    digitalWrite(ledPin3, HIGH);
    lastMatchedPattern = "HELP";
    sendMessageToTelegram("HELP");
  } else if (flexValue1 < 4095) {
    Serial.println("MEDICINE");
    display.setCursor(0, 40);
    display.println("MEDICINE");
    digitalWrite(ledPin1, HIGH);
    lastMatchedPattern = "MEDICINE";
    sendMessageToTelegram("MEDICINE");
  } else if (flexValue2 < 4095) {
    Serial.println("WATER");
    display.setCursor(0, 40);
    display.println("WATER");
    digitalWrite(ledPin2, HIGH);
    lastMatchedPattern = "WATER";
    sendMessageToTelegram("WATER");
  } else {
    digitalWrite(ledPin1, LOW);
    digitalWrite(ledPin2, LOW);
    digitalWrite(ledPin3, LOW);
  }

  display.display();
  delay(1000);
}
